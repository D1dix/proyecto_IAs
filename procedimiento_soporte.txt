"""
Interfaz web para el asistente RAG usando Gradio (bonificación: canal externo +0,5).

Reutiliza la cadena RAG de asistente.py y la expone en una interfaz de chat web.

Uso:
    python ingesta.py        # primero, para indexar (si no lo has hecho)
    python interfaz_web.py   # levanta la web en http://localhost:7860

En Colab, usa demo.launch(share=True) para obtener un enlace público temporal.
"""

import gradio as gr
from asistente import cargar_base_vectorial, crear_cadena_rag

vectorstore = cargar_base_vectorial()
cadena, _ = crear_cadena_rag(vectorstore)


def responder(mensaje, historial):
    """Devuelve la respuesta del sistema RAG a la pregunta del usuario."""
    return cadena.invoke(mensaje)


demo = gr.ChatInterface(
    fn=responder,
    title="Asistente RAG - TechCorp",
    description="Pregunta sobre políticas de RRHH y procedimientos de soporte de TechCorp.",
    examples=[
        "¿Cuántos días de vacaciones tengo al año?",
        "¿Cuál es el procedimiento para reportar una incidencia técnica?",
        "¿Cada cuánto tiempo debo cambiar mi contraseña?",
    ],
)

if __name__ == "__main__":
    # share=True genera un enlace público (útil en Colab).
    demo.launch()
